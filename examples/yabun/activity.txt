' Activity diagram Website Yayasan BUM Pontianak (YABUN)
' Disusun dari routes/web.php dan controller di app/Http/Controllers

=== Login Admin
|Admin|
start
:Membuka halaman /admin/login;
|Sistem|
:Menampilkan form login;
|Admin|
:Mengisi email dan kata sandi;
:Klik tombol Masuk;
|Sistem|
if (Melebihi 5 percobaan per menit?) then (ya)
  :Menolak login sementara;
  stop
else (tidak)
  :Memeriksa email dan kata sandi;
endif
if (Kredensial valid?) then (ya)
  :Regenerasi session;
  :Menampilkan halaman dashboard;
else (tidak)
  :Menampilkan pesan kredensial salah;
  -> Mengisi email dan kata sandi
endif
stop

=== Tambah Donatur
|Admin|
start
:Membuka menu Donatur;
:Klik tombol Tambah Donatur;
|Sistem|
:Menampilkan form donatur dan donasi;
|Admin|
:Mengisi data donatur, kategori, total dan tanggal donasi;
:Klik tombol Simpan;
|Sistem|
:Validasi data;
if (Total donasi minimal Rp 1.000 dan data lengkap?) then (ya)
  fork
    :Menyimpan data donatur;
  fork again
    :Mencatat transaksi pemasukan;
  end fork
  :Menghapus cache ringkasan keuangan;
  :Menampilkan pesan donatur berhasil ditambahkan;
else (tidak)
  :Menampilkan pesan kesalahan;
  -> Mengisi data donatur, kategori, total dan tanggal donasi
endif
stop
