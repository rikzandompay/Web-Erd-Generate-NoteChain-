=== Class Diagram Model
class Authenticatable

class User {
  - id : bigint
  - nama : varchar
  - email : varchar
  - kata_sandi : varchar
  - peran : enum
  - token_ingat : varchar
  + canAccessPanel(panel) : bool
  + setPasswordAttribute(value)
}

class Program {
  - id : bigint
  - nama_program : varchar
  - slug : varchar
  - deskripsi : text
  - ikon : varchar
  - status_aktif : boolean
  + katalogs() : HasMany
}

class Katalog {
  - id : bigint
  - program_id : bigint
  - judul : varchar
  - slug : varchar
  - deskripsi : text
  - gambar : varchar
  - target_penerima : varchar
  - status : enum
  + program() : BelongsTo
  + getGambarUrlAttribute() : string
}

class ProgramCategory {
  - id : bigint
  - name : varchar
  - slug : varchar
  - description : text
  - cover_image : varchar
  + articles() : HasMany
}

class Article {
  - id : bigint
  - program_category_id : bigint
  - title : varchar
  - slug : varchar
  - excerpt : text
  - content : longtext
  - header_image : varchar
  - published_at : timestamp
  - is_published : boolean
  + programCategory() : BelongsTo
  + scopePublished(query)
  + scopeSearch(query, search)
  + generateStaticSitemap()
}

class Donatur {
  - id : bigint
  - nama_donatur : varchar
  - nomor_hp : varchar
  - tipe_donatur : enum
  + financialTransactions() : HasMany
}

class FinancialTransaction {
  - id : bigint
  - type : enum
  - category : enum
  - amount : decimal
  - donor_id : bigint
  - description : text
  - transaction_date : date
  + donor() : BelongsTo
  + scopePemasukan(query)
  + scopePengeluaran(query)
}

class FinanceSummaryService {
  + totalsByCategory(fresh) : array
  + resolveUnit(amount) : string
  + clearCache()
}

class GalleryPhoto {
  - id : bigint
  - title : varchar
  - image_path : varchar
  - category : enum
  - activity_date : date
  - is_featured : boolean
  - sort_order : int
  + getImageUrlAttribute() : string
}

class RekeningBank {
  - id : bigint
  - nama_bank : varchar
  - nomor_rekening : varchar
  - atas_nama : varchar
  - foto_qris : varchar
  - status_aktif : boolean
}

class Pengaturan {
  - id : bigint
  - kunci : varchar
  - nilai : longtext
  - kelompok : varchar
  + ambil(kunci, bawaan) : string
}
